cmake_minimum_required(VERSION 3.22.1)
project(photo_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/llama.cpp")
if(NOT EXISTS "${LLAMA_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Missing vendor/llama.cpp. Run python3 scripts/prepare_vendor.py or build through GitHub Actions.")
endif()
add_subdirectory("${LLAMA_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
add_library(photo_ai SHARED photo_ai.cpp)
target_link_libraries(photo_ai PRIVATE mtmd llama)
if(ANDROID)
    target_link_libraries(photo_ai PRIVATE android log)
    target_link_options(photo_ai PRIVATE "-Wl,-z,max-page-size=16384")
else()
    # The identical JNI engine can be exercised on a Linux CI runner.
    find_package(JNI REQUIRED)
    target_include_directories(photo_ai PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_compile_options(photo_ai PRIVATE -Wall -Wextra)
